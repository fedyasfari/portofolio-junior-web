--- PROGRAM TARIF PARKIR ---
MULAI
INPUT lama parkir dalam jam

JIKA input tidak valid atau <= 0
    TAMPILKAN error
    SELESAI

BULATKAN lama parkir ke atas

HITUNG hari penuh = jam DIV 24
HITUNG sisa jam = jam MOD 24

JIKA sisa jam = 0
    biaya sisa = 0
JIKA sisa jam <= 2
    biaya sisa = 3000
SELAIN ITU
    biaya sisa = 3000 +
                 (sisa jam - 2) * 2000

JIKA biaya sisa > 15000
    biaya sisa = 15000

TOTAL = hari penuh * 15000 + biaya sisa
TAMPILKAN TOTAL
SELESAI


--- KALKULATOR SEDERHANA ---
MULAI
INPUT angka1
INPUT operator
INPUT angka2

JIKA input angka tidak valid
    TAMPILKAN error
    SELESAI

PILIH operator
    "+" : hasil = angka1 + angka2
    "-" : hasil = angka1 - angka2
    "*" : hasil = angka1 * angka2
    "/" :
        JIKA angka2 = 0
            TAMPILKAN error
        SELAIN ITU
            hasil = angka1 / angka2
    LAINNYA:
        TAMPILKAN operator tidak dikenal

JIKA tidak ada error
    TAMPILKAN hasil

SELESAI

--- HITUNG JUMLAH DIGIT ANGKA ---
MULAI

    digit = angka MOD 10
    jumlah = jumlah + digit
    angka = FLOOR(angka / 10)
